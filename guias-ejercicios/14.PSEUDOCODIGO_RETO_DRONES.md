# Pseudocódigo: Dron y App

Este documento es una guía de orientación. No es código Java: cada grupo debe traducirlo y adaptarlo. Si quiere entregarlo a los estudiantes, muévalo a la carpeta `estudiantes/`.

## Clase Dron

```
CLASE Dron

    ATRIBUTOS PRIVADOS
        id        : texto
        modelo    : texto
        bateria   : entero
        enVuelo   : booleano

    CONSTRUCTOR Dron()
        id      <- "SIN-ID"
        modelo  <- "Generico"
        bateria <- 100
        enVuelo <- falso

    CONSTRUCTOR Dron(id, modelo)
        this.id      <- id
        this.modelo  <- modelo
        bateria      <- 100
        enVuelo      <- falso

    CONSTRUCTOR Dron(id, modelo, bateria)
        this.id      <- id
        this.modelo  <- modelo
        this.bateria <- 100            // valor de respaldo
        enVuelo      <- falso
        setBateria(bateria)            // la validación decide si cambia

    getId()       -> retornar id
    getModelo()   -> retornar modelo
    getBateria()  -> retornar bateria
    isEnVuelo()   -> retornar enVuelo

    setBateria(bateria)
        SI bateria >= 0 Y bateria <= 100 ENTONCES
            this.bateria <- bateria
        FIN SI

    resumen() -> texto
        retornar id + " | " + modelo + " | bat=" + bateria + "% | enVuelo=" + enVuelo

    despegar() -> booleano
        SI NO enVuelo Y bateria > 20 ENTONCES
            enVuelo <- verdadero
            retornar verdadero
        FIN SI
        retornar falso

    simularVuelo(minutos) -> entero
        SI NO enVuelo ENTONCES
            retornar 0
        FIN SI
        volados <- 0
        MIENTRAS volados < minutos Y bateria > 20 HACER
            bateria <- bateria - 5
            volados <- volados + 1
        FIN MIENTRAS
        SI bateria <= 20 ENTONCES
            enVuelo <- falso           // aterriza por seguridad
        FIN SI
        retornar volados

FIN CLASE
```

## Clase App

```
CLASE App

    // Métodos auxiliares del checkpoint 6 (misma clase, fuera del principal)

    MÉTODO estático gastar(bateria : entero)
        bateria <- bateria - 50

    MÉTODO estático gastar(d : Dron)
        d.setBateria( d.getBateria() - 50 )

    // PRINCIPAL (se va reemplazando checkpoint a checkpoint)

    PRINCIPAL

        // ---------- Checkpoint 1 ----------
        d <- nuevo Dron()
        imprimir "id: "      + d.getId()
        imprimir "modelo: "  + d.getModelo()
        imprimir "bateria: " + d.getBateria()
        imprimir "enVuelo: " + d.isEnVuelo()

        // ---------- Checkpoint 2 ----------
        a <- nuevo Dron()
        b <- nuevo Dron("D-01", "Falcon")
        c <- nuevo Dron("D-02", "Mavic", 65)
        imprimir a.resumen()
        imprimir b.resumen()
        imprimir c.resumen()

        // ---------- Checkpoint 3 ----------
        d <- nuevo Dron("D-03", "Phantom")
        imprimir "inicial: " + d.getBateria()
        PARA cada valor EN [55, 150, -5, 0]
            d.setBateria(valor)
            imprimir "tras setBateria(" + valor + "): " + d.getBateria()
        e <- nuevo Dron("D-04", "Phantom", 130)
        imprimir "constructor con 130: " + e.getBateria()
        // (puede escribirlo cuatro veces seguidas en vez de usar la lista)

        // ---------- Checkpoint 4 ----------
        d <- nuevo Dron("D-05", "Falcon", 20)
        imprimir "bateria 20, despegar: " + d.despegar()
        d.setBateria(21)
        imprimir "bateria 21, despegar: " + d.despegar()
        imprimir "enVuelo: " + d.isEnVuelo()
        imprimir "despegar de nuevo: " + d.despegar()
        e <- nuevo Dron("D-06", "Mavic", 80)
        imprimir "otro dron, despegar: " + e.despegar()
        imprimir e.resumen()

        // ---------- Checkpoint 5 ----------
        d <- nuevo Dron("D-07", "Falcon")
        imprimir "volar sin despegar: " + d.simularVuelo(5)
        d.despegar()
        minutos <- d.simularVuelo(4)
        imprimir "minutos volados: " + minutos
        imprimir "bateria: " + d.getBateria()
        minutos <- d.simularVuelo(20)
        imprimir "minutos volados: " + minutos
        imprimir "bateria: " + d.getBateria()
        imprimir "enVuelo: " + d.isEnVuelo()
        PARA i DESDE 1 HASTA 3 HACER
            f <- nuevo Dron("F-0" + i, "Falcon", 100 - i * 30)
            imprimir "F-0" + i + " despega: " + f.despegar()
        FIN PARA

        // ---------- Checkpoint 6 ----------
        a <- nuevo Dron("D-08", "Falcon")
        b <- a                              // misma referencia, mismo objeto
        b.setBateria(60)
        imprimir "a: " + a.getBateria()
        imprimir "b: " + b.getBateria()
        imprimir "a == b: " + (a es la misma referencia que b)
        nivel <- 80
        gastar(nivel)                       // se pasa una copia del número
        imprimir "nivel: " + nivel
        gastar(a)                           // se pasa la referencia al objeto
        imprimir "a despues de gastar: " + a.getBateria()
        imprimir "b despues de gastar: " + b.getBateria()
        c <- nuevo Dron("D-08", "Falcon")   // otro objeto, mismos datos
        imprimir "a == c: " + (a es la misma referencia que c)

        // ---------- Checkpoint 7 ----------
        flota <- arreglo de 3 Dron:
                   nuevo Dron("D-01", "Falcon"),
                   nuevo Dron("D-02", "Mavic", 50),
                   nuevo Dron("D-03", "Phantom", 15)

        PARA cada dron d EN flota HACER
            despego <- d.despegar()
            imprimir d.getId() + " despega: " + despego
            SI despego ENTONCES
                minutos <- d.simularVuelo(8)
                imprimir d.getId() + " vuela " + minutos + " minutos"
            FIN SI
            imprimir d.resumen()
        FIN PARA

        enVuelo <- 0
        suma    <- 0
        menor   <- flota[0]
        PARA cada dron d EN flota HACER
            SI d.isEnVuelo() ENTONCES
                enVuelo <- enVuelo + 1
            FIN SI
            suma <- suma + d.getBateria()
            SI d.getBateria() < menor.getBateria() ENTONCES
                menor <- d
            FIN SI
        FIN PARA

        promedio <- suma / (cantidad de drones en flota)   // división con decimales
        imprimir "drones en vuelo: " + enVuelo
        imprimir "bateria promedio: " + promedio con 1 decimal
        imprimir "menor bateria: " + menor.getId()

FIN CLASE
```

Notas para pasar de pseudocódigo a Java:

- `<-` es la asignación (`=` en Java).
- `imprimir` se escribe `System.out.println(...)`.
- `nuevo Dron(...)` es `new Dron(...)`.
- La comparación de referencias `a == b` en Java es literalmente `a == b`.
- En el checkpoint 7, el promedio debe calcularse como `double` (por ejemplo `suma / (double) flota.length`) y formatearse con `String.format(java.util.Locale.US, "%.1f", promedio)`.
